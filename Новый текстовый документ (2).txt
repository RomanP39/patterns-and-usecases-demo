@startuml
class Icon{
draw(x,y)
}
class FileIcon{
+file_type
FileIcon(Icon)
draw(x,y)
}
class IconFactory{
-icons{}
{static} get_icon(file_type)
}
class VideoPlayer{
play(file)
}
class AudioPlayer{
play(file)
}
class MediaFacade{
-video: VideoPlayer
-audio: AudioPlayer
+play_media(file)
}
Icon <|-- FileIcon
IconFactory o-- Icon

MediaFacade -- VideoPlayer
MediaFacade --AudioPlayer
MediaFacade --> FileIcon
@enduml



@startuml

@enduml
